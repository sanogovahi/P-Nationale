export const queryAsync = (sql, params = []) => {
  return new Promise((resolve, reject) => {
    const db = getDatabase();
    db.all(sql, params, (err, rows) => {
      if (err) reject(err);
      else resolve(rows);
    });
  });
};

export const runAsync = (sql, params = []) => {
  return new Promise((resolve, reject) => {
    const db = getDatabase();
    db.run(sql, params, function(err) {
      if (err) reject(err);
      else resolve({ id: this.lastID, changes: this.changes });
    });
  });
};

document.addEventListener("DOMContentLoaded", function () {

  const sliders = document.querySelectorAll(".ad-container");

  sliders.forEach(container => {
    const track = container.querySelector(".ad-track");
    const slides = container.querySelectorAll(".ad-slide");

    if (!track || slides.length === 0) return;

    let index = 0;
    let direction = 1;

    function update() {
      track.style.transform = "translateX(-" + (index * 100) + "%)";
    }

    function move() {
      index += direction;

      if (index >= slides.length - 1) {
        index = slides.length - 1;
        direction = -1;
      } 
      else if (index <= 0) {
        index = 0;
        direction = 1;
      }

      update();
    }

    let interval = setInterval(move, 3000);

    /* pause au survol */
    container.addEventListener("mouseenter", () => clearInterval(interval));
    container.addEventListener("mouseleave", () => {
      interval = setInterval(move, 3000);
    });

  });

});
